-- model_config : Extra inputs are not permitted 

from pydantic import BaseModel, ConfigDict

class Student(BaseModel):

    model_config = ConfigDict(
        extra="forbid"
    )

    name: str
    age: int

input : Student(
    name="Rahul",
    age=22,
    city="Ahmedabad"
)

-- Custom validation: field_validator, model_validator, mode="before" / "after"

--Field validator : Field validator : use for validating one filed 

from pydantic import BaseModel, field_validator

class User(BaseModel):
    name: str

    @field_validator("name")
    @classmethod
    def remove_spaces(cls, value):
        return value.strip()

input : User(name="  Bansari  ")

output : name='Bansari'

--Create two models.

from typing import Literal

class Success(BaseModel):

    type:Literal["success"]
    answer:str

class Error(BaseModel):

    type:Literal["error"]
    message:str

Combine them.

from typing import Union, Annotated
from pydantic import Field

Response = Annotated[
    Union[Success, Error],
    Field(discriminator="type")
]

-- Boundary design: the model's raw response type must never be your domain type

One of the most important production concepts.

Never trust raw LLM output.

--Raw LLM : {
"career":"AI Engineer",
"confidence":0.91
}

--llm model : class LLMResponse(BaseModel):

    career:str
    confidence:float

--Domain model : class CareerRecommendation(BaseModel):

    role:str
    score:float

--Mapping : CareerRecommendation(

role=response.career,

score=response.confidence
)
This separation keeps your business logic independent of the LLM's output format.

-- Buisness Rule violations 

The data is structurally valid but breaks your application's rules.

Example

category="Hotel"

confidence=None

Both fields have valid types, but your rule says confidence is required whenever a category is assigned.

This is where model_validator is commonly used, or you may raise a custom exception.
